# 分布式推理协调器 - 详细工作流程说明

## 一、整体架构概览

### 1.1 系统组件

```
┌─────────────────────────────────────────────────────────────────┐
│                        用户电脑 A                                │
│  ┌─────────────┐    ┌─────────────┐    ┌─────────────┐        │
│  │  前端界面   │───▶│  协调器     │───▶│  本地推理   │        │
│  │             │    │ Coordinator │    │  Engine     │        │
│  └─────────────┘    └──────┬──────┘    └─────────────┘        │
│                            │                                    │
│                     iroh P2P 网络                              │
└────────────────────────────┼───────────────────────────────────┘
                             │
              ┌──────────────┼──────────────┐
              │              │              │
              ▼              ▼              ▼
┌─────────────────┐ ┌─────────────────┐ ┌─────────────────┐
│   电脑 B        │ │   电脑 C        │ │   电脑 D        │
│ ┌─────────────┐ │ │ ┌─────────────┐ │ │ ┌─────────────┐ │
│ │ 分片 1      │ │ │ │ 分片 2      │ │ │ │ 分片 3      │ │
│ │ Layers 0-10 │ │ │ │ Layers 11-20│ │ │ │ Layers 21-30│ │
│ └─────────────┘ │ │ └─────────────┘ │ │ └─────────────┘ │
└─────────────────┘ └─────────────────┘ └─────────────────┘
```

### 1.2 核心概念

| 概念 | 说明 |
|------|------|
| **协调器** | 负责管理整个分布式推理流程，决定哪个节点执行哪个分片 |
| **分片** | 模型被切分成多个部分，每个部分可以在不同节点上运行 |
| **中间结果** | 一个分片的输出，作为下一个分片的输入 |
| **iroh P2P** | 底层通信协议，用于节点间数据传输 |

## 三、数据流详解

### 3.1 消息格式

```rust
// 执行分片请求
{
    "type": "ExecuteShard",
    "shard_id": "shard_1",
    "input_data": [0.1, 0.2, 0.3, ...],  // 张量数据
    "metadata": {
        "model_id": "llama-3.2-1b",
        "layer_start": 0,
        "layer_end": 10
    }
}

// 执行结果响应
{
    "type": "ExecutionResult",
    "shard_id": "shard_1",
    "output_data": [0.15, 0.25, 0.35, ...],  // 输出张量
    "metrics": {
        "execution_time_ms": 150,
        "gpu_memory_used_mb": 2048
    }
}
```

### 3.2 分片表结构

```rust
// 协调器维护的分片表
{
    "model_llama_3.2_1b": {
        "shard_0": {

            "layer_range": [0, 10],
            "size_mb": 350,
            "status": "ready"
        },
        "shard_1": {
            "node_id": "node_c_def456",
            "layer_range": [11, 20],
            "size_mb": 350,
            "status": "ready"
        },
        "shard_2": {
            "node_id": "node_d_ghi789",
            "layer_range": [21, 30],
            "size_mb": 350,
            "status": "ready"
        }
    }
}
```

## 四、AI 决策的作用

### 4.1 切分决策

AI 决策引擎会分析：

```rust
// 输入给 AI 的上下文
{
    "available_nodes": [
        {"id": "node_b", "gpu_memory_gb": 8, "latency_ms": 10},
        {"id": "node_c", "gpu_memory_gb": 12, "latency_ms": 15},
        {"id": "node_d", "gpu_memory_gb": 6, "latency_ms": 20}
    ],
    "model_info": {
        "total_layers": 30,
        "total_params": "1B",
        "memory_required_gb": 4
    }
}

// AI 返回的决策
{
    "decision_type": "split_model",
    "action": "按节点算力分配层数",
    "parameters": {
        "node_b": {"layers": "0-10", "reason": "低延迟，适合首层"},
        "node_c": {"layers": "11-20", "reason": "高内存，适合中间层"},
        "node_d": {"layers": "21-30", "reason": "剩余层"}
    }
}
```

### 4.2 运行时调度

如果某个节点失败，AI 可以重新调度：

```rust
// AI 决策
{
    "decision_type": "reschedule",
    "action": "将分片 2 从失败的节点 C 迁移到节点 B",
    "reasoning": "节点 C 无响应，节点 B 有足够内存"
}
```

## 六、容错机制

### 6.1 节点失败处理

```
节点 C 失败
     │
     ▼
┌───────────────────────────────────────┐
│ 1. 检测失败                            │
│    - 心跳超时                          │
│    - 执行请求无响应                    │
└───────────────────────────────────────┘
     │
     ▼
┌───────────────────────────────────────┐
│ 2. AI 决策重新调度                     │
│    - 选择替代节点                      │
│    - 传输分片到新节点                  │
└───────────────────────────────────────┘
     │
     ▼
┌───────────────────────────────────────┐
│ 3. 恢复执行                            │
│    - 从上一个成功的分片继续            │
│    - 使用缓存的中间结果                │
└───────────────────────────────────────┘
```

### 6.2 中间结果缓存

```rust
// 缓存结构
{
    "task_123": {
        "shard_0_result": [0.15, 0.25, ...],  // 分片 0 的输出
        "shard_1_result": null,                // 分片 1 尚未完成
        "timestamp": 1707892800
    }
}
```

## 八、总结

```
1. 节点发现 → 2. 模型下载 → 3. 模型切分 → 4. 分片分发 → 5. 分布式推理
```
